


        

      
    
 import 'package:flutter/material.dart';
import 'package:url_launcher/url_launcher.dart';

void main() {
  runApp(const BallaghApp());
}

class BallaghApp extends StatelessWidget {
  const BallaghApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'بَلِّغْ - Ballagh',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        primaryColor: const Color(0xFF0F1E36),
        scaffoldBackgroundColor: const Color(0xFF1A2A40),
        colorScheme: ColorScheme.fromSeed(
          seedColor: const Color(0xFF0F1E36),
          secondary: const Color(0xFFD32F2F),
        ),
      ),
      home: const MainLayout(),
    );
  }
}

class MainLayout extends StatefulWidget {
  const MainLayout({super.key});

  @override
  State<MainLayout> createState() => _MainLayoutState();
}

class _MainLayoutState extends State<MainLayout> {
  int _currentIndex = 0;

  final List<Widget> _screens = [
    const DirectReportScreen(),
    const GroupReportScreen(),
    const InfoScreen(),
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'بَلِّغْ',
          style: TextStyle(fontWeight: FontWeight.bold, color: Colors.white, fontSize: 24),
        ),
        centerTitle: true,
        backgroundColor: const Color(0xFF0F1E36),
        elevation: 4,
      ),
      body: _screens[_currentIndex],
      bottomNavigationBar: BottomNavigationBar(
        currentIndex: _currentIndex,
        backgroundColor: const Color(0xFF0F1E36),
        selectedItemColor: Colors.redAccent,
        unselectedItemColor: Colors.white70,
        onTap: (index) {
          setState(() {
            _currentIndex = index;
          });
        },
        items: const [
          BottomNavigationBarItem(
            icon: Icon(Icons.report_problem),
            label: 'تبليغ مباشر',
          ),
          BottomNavigationBarItem(
            icon: Icon(Icons.groups),
            label: 'تبليغ جماعي',
          ),
          BottomNavigationBarItem(
            icon: Icon(Icons.security),
            label: 'توعية أمنية',
          ),
        ],
      ),
    );
  }
}

class DirectReportScreen extends StatelessWidget {
  const DirectReportScreen({super.key});

  Future<void> _launchURL(String urlString) async {
    final Uri url = Uri.parse(urlString);
    if (!await launchUrl(url, mode: LaunchMode.externalApplication)) {
      throw Exception('Could not launch $urlString');
    }
  }

  @override
  Widget build(BuildContext context) {
    return ListView(
      padding: const EdgeInsets.all(16.0),
      children: [
        const Text(
          'اختر المنصة ونوع المحتوى للتبليغ الفوري:',
          style: TextStyle(fontSize: 18, color: Colors.white70),
          textAlign: TextAlign.right,
        ),
        const SizedBox(height: 16),
        _buildPlatformSection(
          title: 'إنستغرام - Instagram',
          icon: Icons.camera_alt,
          color: const Color(0xFFE1306C),
          buttons: [
            _buildReportButton('🔞 محتوى خادش للحياء / +18', 'https://help.instagram.com/contact/383679321740945', isUrgent: true),
            _buildReportButton('⚠️ ابتزاز أو تهديد مالي', 'https://help.instagram.com/contact/1681794611860555'),
            _buildReportButton('👤 انتحال شخصية أو نصب', 'https://help.instagram.com/446663175382270'),
          ],
        ),
        const SizedBox(height: 24),
        _buildPlatformSection(
          title: 'تيك توك - TikTok',
          icon: Icons.music_note,
          color: Colors.black,
          buttons: [
            _buildReportButton('🔞 محتوى البالغين و +18', 'https://www.tiktok.com/legal/report/feedback', isUrgent: true),
            _buildReportButton('⚠️ انتهاك إرشادات المجتمع والنصب', 'https://support.tiktok.com/ar/safety-hc/report-a-problem'),
          ],
        ),
      ],
    );
  }

  Widget _buildPlatformSection({required String title, required IconData icon, required Color color, required List<Widget> buttons}) {
    return Card(
      color: const Color(0xFF1E324F),
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      child: Padding(
        padding: const EdgeInsets.all(12.0),
        child: Column(
          children: [
            Row(
              mainAxisAlignment: MainAxisAlignment.end,
              children: [
                Text(title, style: const TextStyle(fontSize: 18, fontWeight: FontWeight.bold, color: Colors.white)),
                const SizedBox(width: 10),
                Icon(icon, color: color, size: 28),
              ],
            ),
            const Divider(color: Colors.white24, height: 20),
            ...buttons,
          ],
        ),
      ),
    );
  }

  Widget _buildReportButton(String label, String url, {bool isUrgent = false}) {
    return Container(
      width: double.infinity,
      margin: const EdgeInsets.symmetric(vertical: 6),
      child: ElevatedButton(
        style: ElevatedButton.styleFrom(
          backgroundColor: isUrgent ? const Color(0xFFD32F2F) : const Color(0xFF0F1E36),
          padding: const EdgeInsets.symmetric(vertical: 14),
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
        ),
        onPressed: () => _launchURL(url),
        child: Text(
          label,
          style: const TextStyle(color: Colors.white, fontSize: 16, fontWeight: FontWeight.bold),
          textAlign: TextAlign.center,
        ),
      ),
    );
  }
}
